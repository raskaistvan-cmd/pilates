function submitRegistration(timeSlot) { const nameInput = document.querySelector(`#timeslot-${timeSlot} input[placeholder*="Név"]`); const emailInput = document.querySelector(`#timeslot-${timeSlot} input[placeholder*="Email"]`); const name = nameInput?.value?.trim(); if (!name) { alert('Kérlek, írj be egy nevet!'); return; } const email = emailInput?.value?.trim() || ''; const url = APPS_SCRIPT_URL + '?action=add&timeSlot=' + encodeURIComponent(timeSlot) + '&name=' + encodeURIComponent(name) + '&email=' + encodeURIComponent(email); fetch(url) .then(response => response.json()) .then(data => { console.log('✅ Response:', data); document.querySelector(`#timeslot-${timeSlot} input[placeholder*="Név"]`).value = ''; document.querySelector(`#timeslot-${timeSlot} input[placeholder*="Email"]`).value = ''; loadData(); showStatus('✅ Regisztráció sikeres!', 'success'); }) .catch(error => { console.error('Hiba:', error); loadData(); }); }
