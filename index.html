function submitRegistration(timeSlot) { const nameInput = document.querySelector(`#timeslot-${timeSlot} input[placeholder*="Név"]`); const emailInput = document.querySelector(`#timeslot-${timeSlot} input[placeholder*="Email"]`); const name = nameInput?.value?.trim(); if (!name) { alert('Kérlek, írj be egy nevet!'); return; } const email = emailInput?.value?.trim() || ''; const formData = new FormData(); formData.append('action', 'add'); formData.append('timeSlot', timeSlot); formData.append('name', name); formData.append('email', email); fetch(APPS_SCRIPT_URL, { method: 'POST', body: formData, mode: 'no-cors' }) .then(() => { document.querySelector(`#timeslot-${timeSlot} input[placeholder*="Név"]`).value = ''; document.querySelector(`#timeslot-${timeSlot} input[placeholder*="Email"]`).value = ''; loadData(); showStatus('✅ Regisztráció sikeres!', 'success'); }) .catch(error => { console.log('Fetch error (normal):', error); loadData(); }); }
